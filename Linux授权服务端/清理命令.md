LicenseProtector Linux 服务端清理命令

注意：以下命令会删除 Agent、运行密钥、配置和 PHP Loader，不备份；不会删除项目目录中的 PHP 加密文件。请确认要彻底卸载后，以 root 身份整段复制执行。

```bash
(
set -Eeuo pipefail
if [[ $EUID -ne 0 ]]; then
  echo '请在 root 终端执行' >&2
  exit 1
fi

echo '== 停止并删除 LicenseProtector 服务 =='
systemctl disable --now license-agent.service 2>/dev/null || true
rm -f /etc/systemd/system/license-agent.service
systemctl daemon-reload

echo '== 删除 Agent、配置、密钥和运行目录 =='
rm -rf /opt/license-protector-agent
rm -rf /etc/license-protector
rm -rf /run/license-protector

echo '== 删除 PHP Loader 配置 =='
find /www/server/php -type f -path '*/etc/*.ini' -exec sed -i '/license_loader\.so/d' {} \; 2>/dev/null

echo '== 删除 PHP Loader 文件 =='
find /www/server/php -type f -name 'license_loader.so' -delete 2>/dev/null

echo '== 删除 Agent 用户和用户组（存在才删除） =='
userdel license-agent 2>/dev/null || true
groupdel license-agent 2>/dev/null || true

echo '== 清理完成：检查残留 =='
systemctl status license-agent.service --no-pager 2>/dev/null || true
find /www/server/php /opt /etc /run \
  \( -name 'license_loader.so' -o -name 'license-agent.service' -o -path '*/license-protector*' \) \
  -print 2>/dev/null
)
```
